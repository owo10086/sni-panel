import assert from 'node:assert/strict';
import fs from 'node:fs';
import net from 'node:net';
import tls from 'node:tls';
import path from 'node:path';
import os from 'node:os';
import { spawn, spawnSync } from 'node:child_process';
import { performance } from 'node:perf_hooks';
const panel = process.env.VERIFY_PANEL_URL;
const directory = fs.mkdtempSync(path.join(os.tmpdir(),'ticket03-live-'));
const control = path.join(directory,'fxp.sock');
async function runtimeStatus() {
  return new Promise((resolve,reject)=>{
    const socket=net.connect(control);let input='';
    socket.setTimeout(2000,()=>socket.destroy(new Error('Status timeout')));
    socket.once('error',reject);
    socket.once('connect',()=>socket.write(JSON.stringify({operation:'status'})+'\n'));
    socket.on('data',chunk=>{input+=chunk;if(input.includes('\n')){const status=JSON.parse(input.trim());socket.destroy();assert.equal(status.ok,true);resolve(status);}});
  });
}
async function api(action, body = {}) {
  if (fs.existsSync(control) && ['snapshot','reset-quota'].includes(action)) {
    const status=await runtimeStatus();
    body.localState={rules:status.sniRoutes.map(route=>({...route,port:config.listenPort,forwardType:'forwardx',protocol:'tcp',sniRouteVersion:status.version,sourceAllowIps:[],ready:true})),tunnels:[],services:[]};
  }
  const response = await fetch(panel + '/verification/' + action,{method:'POST',headers:{'content-type':'application/json'},body:JSON.stringify(body)});
  const payload = await response.json();
  assert.equal(response.status,200,JSON.stringify(payload));
  return payload;
}
function execute(binary,args) {
  const result = spawnSync(binary,args,{encoding:'utf8',timeout:15000});
  assert.equal(result.status,0,result.stderr || String(result.error));
}
async function poll(action,predicate) {
  const deadline = Date.now()+35000;
  let value;
  do {
    value = await api(action);
    if (predicate(value)) return value;
    await new Promise(resolve=>setTimeout(resolve,100));
  } while(Date.now()<deadline);
  assert.fail('Timed out polling '+action+': '+JSON.stringify(value));
}
function connect(sni,timeout=25000) {
  return new Promise((resolve,reject)=>{
    const socket = tls.connect({host:'127.0.0.1',port:config.listenPort,servername:sni,rejectUnauthorized:false});
    socket.setTimeout(timeout,()=>socket.destroy(new Error('TLS timeout')));
    socket.once('error',reject);
    socket.once('secureConnect',()=>{socket.on('error',()=>{});resolve(socket);});
  });
}
function exchange(socket,payload) {
  assert.equal(socket.destroyed,false,'Echo requires a live TLS connection');
  return new Promise((resolve,reject)=>{
    const parts=[];
    let received=0;
    const error=err=>{cleanup();reject(err);};
    const close=()=>error(new Error('TLS closed before full echo'));
    const data=part=>{parts.push(part);received+=part.length;if(received>=payload.length){cleanup();const actual=Buffer.concat(parts);assert.deepEqual(actual,payload);resolve(actual.length);}};
    function cleanup(){socket.off('data',data);socket.off('error',error);socket.off('close',close);}
    socket.on('data',data);socket.once('error',error);socket.once('close',close);
    socket.write(payload);
  });
}
async function request(sni,payload=Buffer.from('probe')) {
  const socket = await connect(sni);
  try {return await exchange(socket,payload);} finally {socket.destroy();}
}
function update(next) {
  return new Promise((resolve,reject)=>{
    const socket=net.connect(control);
    let input='';
    socket.setTimeout(2000,()=>socket.destroy(new Error('Control timeout')));
    socket.once('error',reject);
    socket.once('connect',()=>socket.write(JSON.stringify({version:next.sniRouteVersion,sniRoutes:next.sniRoutes})+'\n'));
    socket.on('data',chunk=>{input+=chunk;if(input.includes('\n')){const result=JSON.parse(input.trim());socket.destroy();assert.equal(result.ok,true,result.error);resolve(result);}});
  });
}

const config = {...initial.config,panelUrl:panel,token:'exit-token',controlSocketPath:control};
const route = config.sniRoutes.find(item=>item.sni==='api.example.com');
assert.equal(route.limitIn,6250000);
assert.equal(route.limitOut,6250000);
assert.equal(route.maxConnections,7);
const addresses = new Set(config.sniRoutes.map(item=>item.targetIp));
for (const address of addresses) execute('ip',['address','add',address+'/32','dev','lo']);
execute('openssl',['req','-x509','-newkey','rsa:2048','-nodes','-keyout',path.join(directory,'key.pem'),'-out',path.join(directory,'cert.pem'),'-subj','/CN=ticket03-live','-days','1']);
