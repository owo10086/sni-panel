const cookies={};
async function rpc(path,input,user='admin',query=false){
 const url='https://127.0.0.1:3000/api/trpc/'+path+(query?'?input='+encodeURIComponent(JSON.stringify({json:input})): '');
 const r=await fetch(url,{method:query?'GET':'POST',headers:{'content-type':'application/json',...(cookies[user]?{cookie:cookies[user]}:{})},...(query?{}:{body:JSON.stringify({json:input})}),signal:AbortSignal.timeout(15000)});
 const raw=await r.json();
 if(raw.error)throw new Error(raw.error.json?.message||JSON.stringify(raw.error));
 const setCookie=r.headers.get('set-cookie');if(path==='auth.login'){assert.ok(setCookie);cookies[user]=setCookie.split(';')[0];}
 return raw.result.data.json;
}
